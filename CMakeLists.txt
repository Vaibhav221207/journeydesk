cmake_minimum_required(VERSION 3.20)
project(JourneyDesk VERSION 2.0.0 LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Threads REQUIRED)

add_library(journey_key src/desk_key.cpp)
target_include_directories(journey_key PUBLIC include)
target_link_libraries(journey_key PUBLIC Threads::Threads)
add_library(journey_engine src/engine.cpp)
target_include_directories(journey_engine PUBLIC include vendor)
target_link_libraries(journey_engine PUBLIC Threads::Threads)
add_library(journey_web src/web.cpp)
target_include_directories(journey_web PUBLIC include vendor)
target_link_libraries(journey_web PUBLIC journey_engine)

if(WIN32)
    target_compile_definitions(journey_web PUBLIC _WIN32_WINNT=0x0A00)
    target_link_libraries(journey_web PUBLIC ws2_32 crypt32)
endif()
if(MSVC)
    target_compile_options(journey_key PRIVATE /W4 /utf-8)
    target_compile_options(journey_engine PRIVATE /W4 /utf-8)
    target_compile_options(journey_web PRIVATE /W4 /utf-8)
else()
    target_compile_options(journey_key PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(journey_engine PRIVATE -Wall -Wextra -Wpedantic)
    target_compile_options(journey_web PRIVATE -Wall -Wextra -Wpedantic)
endif()

add_executable(journeydesk src/main.cpp)
target_link_libraries(journeydesk PRIVATE journey_web journey_key)
if(WIN32)
    target_link_libraries(journeydesk PRIVATE shell32)
endif()
if(MINGW)
    target_link_options(journeydesk PRIVATE -static -s)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(key_tests tests/desk_key_tests.cpp)
    target_link_libraries(key_tests PRIVATE journey_key)
    add_test(NAME desk_key_resolution COMMAND key_tests)
    add_executable(engine_tests tests/engine_tests.cpp)
    target_link_libraries(engine_tests PRIVATE journey_engine)
    add_test(NAME reservation_engine COMMAND engine_tests)
    add_executable(web_tests tests/web_tests.cpp)
    target_link_libraries(web_tests PRIVATE journey_web)
    add_test(NAME http_transport COMMAND web_tests)
    set_tests_properties(desk_key_resolution reservation_engine http_transport PROPERTIES TIMEOUT 90)
endif()
